코드 예시#

인라인 코드는 console.log("Hello World!") 이렇게 표시됩니다.

// JavaScript 코드 예시
function greet(name) {
    const message = `안녕하세요, ${name}님!`;
    console.log(message);
    return message;
}

greet("이우진");

# Python 코드 예시
def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n-1) + fibonacci(n-2)

for i in range(10):
    print(fibonacci(i))
